package rep02;

import java.io.IOException;
import java.net.URI;
import java.net.http.*;
import java.nio.charset.StandardCharsets;
import java.util.Base64;
import java.util.function.Consumer;
import java.util.stream.Stream;

public class IpfsPubSub {
    private final String api;
    private final HttpClient http = HttpClient.newHttpClient();

    public IpfsPubSub(String api) {
        this.api = api;
    }

    static String toMultibase(String s) {
        return "u" + Base64.getUrlEncoder().withoutPadding()
                .encodeToString(s.getBytes(StandardCharsets.UTF_8));
    }

    static String fromMultibase(String s) {
        if (!s.startsWith("u")) {
            throw new IllegalArgumentException("Codificação multibase não suportada: " + s);
        }
        return new String(Base64.getUrlDecoder().decode(s.substring(1)), StandardCharsets.UTF_8);
    }

    public void publish(String topic, String message) throws IOException, InterruptedException {
        String boundary = "rep01-" + System.nanoTime();
        String body = "--" + boundary + "\r\n"
                + "Content-Disposition: form-data; name=\"file\"; filename=\"msg\"\r\n"
                + "Content-Type: application/octet-stream\r\n\r\n"
                + message + "\r\n"
                + "--" + boundary + "--\r\n";
        HttpRequest req = HttpRequest.newBuilder(URI.create(api + "/pubsub/pub?arg=" + toMultibase(topic)))
                .header("Content-Type", "multipart/form-data; boundary=" + boundary)
                .POST(HttpRequest.BodyPublishers.ofString(body, StandardCharsets.UTF_8))
                .build();
        HttpResponse<String> resp = http.send(req, HttpResponse.BodyHandlers.ofString());
        if (resp.statusCode() != 200) {
            throw new IOException("pub falhou (" + resp.statusCode() + "): " + resp.body());
        }
    }

    public void subscribe(String topic, Consumer<String> onMessage) throws IOException, InterruptedException {
        HttpRequest req = HttpRequest.newBuilder(URI.create(api + "/pubsub/sub?arg=" + toMultibase(topic)))
                .POST(HttpRequest.BodyPublishers.noBody())
                .build();
        HttpResponse<Stream<String>> resp = http.send(req, HttpResponse.BodyHandlers.ofLines());
        if (resp.statusCode() != 200) {
            throw new IOException("sub falhou (" + resp.statusCode() + ")");
        }
        resp.body()
                .filter(json -> json.contains("\"data\""))
                .forEach(json -> onMessage.accept(fromMultibase(extractField(json, "data"))));
    }

    static String extractField(String json, String field) {
        String key = "\"" + field + "\":\"";
        int start = json.indexOf(key);
        if (start < 0) {
            throw new IllegalArgumentException("Campo '" + field + "' ausente: " + json);
        }
        start += key.length();
        return json.substring(start, json.indexOf('"', start));
    }
}

public class PartitionLeader {
    private final int k;
    private final int n;
    private final int port;

    public PartitionLeader(int k, int n, String host, String api) throws Exception {
        this.k = k;
        this.n = n;
        this.port = Wire.PORTO_BASE + k;
        this.file = new RecordFile("lider-p" + k + ".txt");
        this.seq = file.lastSeq();

        IpfsPubSub pubsub = new IpfsPubSub(api);

        // Lança o servidor TCP de recuperação numa thread separada
        new Thread(this::startTcpServer).start();

        System.out.println("Líder p" + k + " (N=" + n + ") pronto.");
        System.out.println(" - Tópico de entrada: " + Wire.topicIn(k));
        System.out.println(" - Servidor TCP de recuperação à escuta em: " + host + ":" + port);

        // Subscreve o tópico de entrada da partição k no PubSub
        pubsub.subscribe(Wire.topicIn(k), msg -> {
            try {
                String trimmedMsg = msg != null ? msg.trim() : "";
                String[] parts = trimmedMsg.split(";");

                // Validação estrita do formato: rejeita se não tiver 3 campos ou não começar por "IN"
                if (parts.length != 3 || !parts[0].equals("IN")) {
                    System.out.println("REJEITADO: " + trimmedMsg);
                    return;
                }

                String sensor = parts[1];
                double temp;

                // Validação da temperatura: rejeita se não for um número válido
                try {
                    temp = Double.parseDouble(parts[2]);
                } catch (NumberFormatException e) {
                    System.out.println("REJEITADO: " + trimmedMsg);
                    return;
                }

                SensorRecord r;
                synchronized (this) {
                    r = SensorRecord.now(++seq, sensor, temp);
                    file.append(r);
                }

                String reg = Wire.reg(host, port, r);
                pubsub.publish(Wire.topicData(k), reg);
                System.out.println("p" + k + " Registado e publicado: " + r.toLine());

            } catch (Exception e) {
                System.out.println("REJEITADO: " + msg + " (Erro: " + e.getMessage() + ")");
            }
        });
    }

    private void startTcpServer() {
        try (ServerSocket server = new ServerSocket(port)) {
            while (true) {
                Socket socket = server.accept();
                handleRecovery(socket);
            }
        } catch (IOException e) {
            System.out.println("Erro no servidor TCP p" + k + ": " + e.getMessage());
        }
    }

    private void handleRecovery(Socket socket) {
        try (BufferedReader in = new BufferedReader(new InputStreamReader(socket.getInputStream(), StandardCharsets.UTF_8));
             PrintWriter out = new PrintWriter(new OutputStreamWriter(socket.getOutputStream(), StandardCharsets.UTF_8), true)) {

            socket.setSoTimeout(Wire.T_TCP_MS);
            String req = in.readLine();

            if (req == null) return;
            String[] p = req.trim().split("\\s+");

            if (p.length != 3 || !p[0].equals("PEDIDO")) {
                out.println("ERRO Pedido mal formado");
                return;
            }

            long from, to;
            try {
                from = Long.parseLong(p[1]);
                to = Long.parseLong(p[2]);
            } catch (NumberFormatException e) {
                out.println("ERRO Valores numéricos inválidos");
                return;
            }

            if (from > to) {
                out.println("ERRO Intervalo inválido (from > to)");
                return;
            }

            long top;
            synchronized (this) {
                top = Math.min(to, seq);
            }

            // Lê o intervalo de registos de lider-p<k>.txt
            java.util.List<SensorRecord> records = file.range(from, top);

            // Log de recuperação ajustado ao requisito observável
            System.out.println("RECUPERAÇÃO: " + from + ".." + top + " para " + socket.getRemoteSocketAddress() + " (" + records.size() + " registos)");

            for (SensorRecord r : records) {
                out.println(r.toLine());
            }
            out.println("FIM");

        } catch (Exception e) {
            System.out.println("Erro na recuperação TCP p" + k + ": " + e.getMessage());
        }
    }

    public static void main(String[] args) {
        if (args.length < 2) {
            System.out.println("Erro: Argumentos insuficientes.");
            System.out.println("Uso: java rep02.PartitionLeader <k> <N> [endereço] [api]");
            return;
        }

        int k, n;
        try {
            k = Integer.parseInt(args[0]);
            n = Integer.parseInt(args[1]);
        } catch (NumberFormatException e) {
            System.out.println("Erro: Os parâmetros <k> e <N> têm de ser valores numéricos inteiros.");
            System.out.println("Uso: java rep02.PartitionLeader <k> <N> [endereço] [api]");
            return;
        }

        if (n <= 0 || k < 0 || k >= n) {
            System.out.println("Erro: Partição inválida (k=" + k + " para N=" + n + "). O valor de k deve estar compreendido entre 0 e " + (n - 1) + ".");
            System.out.println("Uso: java rep02.PartitionLeader <k> <N> [endereço] [api]");

package rep02;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.ArrayList;
import java.util.List;

public class RecordFile {
    private final Path path;

    public RecordFile(String fileName) {
        this.path = Paths.get(fileName);
    }

    public synchronized List<SensorRecord> range(long from, long to) throws IOException {
        List<SensorRecord> out = new ArrayList<>();
        if (!Files.exists(path)) {
            return out;
        }
        for (String line : Files.readAllLines(path, StandardCharsets.UTF_8)) {
            if (line.isBlank()) {
                continue;
            }
            SensorRecord r = SensorRecord.fromLine(line);
            if (r.getSeq() >= from && r.getSeq() <= to) {
                out.add(r);
            }
        }
        return out;
    }

    // acrescenta uma linha ao fim do ficheiro (cria-o se não existir)
    public synchronized void append(SensorRecord r) throws IOException {
        Files.writeString(path, r.toLine() + System.lineSeparator(), StandardCharsets.UTF_8,
                StandardOpenOption.CREATE, StandardOpenOption.APPEND);
    }

    // número de sequência do último registo guardado (0 se o ficheiro não existir)
    public synchronized long lastSeq() throws IOException {
        if (!Files.exists(path)) {
            return 0;
        }
        long last = 0;
        for (String line : Files.readAllLines(path, StandardCharsets.UTF_8)) {
            if (!line.isBlank()) {
                last = SensorRecord.fromLine(line).getSeq();
            }
        }
        return last;
    }
}

        return m.substring(0, i);

    public static String body(String msg) {
        String m = msg.trim();
        return m.substring(m.indexOf(';') + 1);

    public static String[] regParts(String msg) {
        String[] f = msg.trim().split(";", 4);

        IpfsPubSub pubsub = new IpfsPubSub(api);

            System.out.println("Produtor pronto (Parte A). Formato: <sensor> <temperatura> | 'rajada <n> <s>' | 'sair'");

            while (true) {
                System.out.print("> ");
                if (!sc.hasNextLine()) break;
                String line = sc.nextLine().trim();
                if (line.isEmpty()) continue;
                if (line.equalsIgnoreCase("sair")) break;

                if (line.toLowerCase().startsWith("rajada")) {
                    String[] parts = line.split("\\s+");
                    int n = parts.length > 1 ? Integer.parseInt(parts[1]) : 5;
                    int sMax = parts.length > 2 ? Integer.parseInt(parts[2]) : 1;

                        double temp = 15 + rand.nextDouble() * 15;
                        
                        String msg = Wire.in(sensor, temp);
                        pubsub.publish(Wire.topicIn(0), msg);
                    }
                    long tEnd = System.currentTimeMillis();

                    System.out.println("Rajada concluída.");

                + "Content-Disposition: form-data; name=\"file\"; filename=\"msg\"\r\n"
                + "Content-Type: application/octet-stream\r\n\r\n"
                + message + "\r\n"
                + "--" + boundary + "--\r\n";
        HttpRequest req = HttpRequest.newBuilder(URI.create(api + "/pubsub/pub?arg=" + toMultibase(topic)))
                .header("Content-Type", "multipart/form-data; boundary=" + boundary)
                .POST(HttpRequest.BodyPublishers.ofString(body, StandardCharsets.UTF_8))
                .build();
        HttpResponse<String> resp = http.send(req, HttpResponse.BodyHandlers.ofString());
        if (resp.statusCode() != 200) {
            throw new IOException("pub falhou (" + resp.statusCode() + "): " + resp.body());
        }
    }

    public void subscribe(String topic, Consumer<String> onMessage) throws IOException, InterruptedException {
        HttpRequest req = HttpRequest.newBuilder(URI.create(api + "/pubsub/sub?arg=" + toMultibase(topic)))
                .POST(HttpRequest.BodyPublishers.noBody())
                .build();
        HttpResponse<Stream<String>> resp = http.send(req, HttpResponse.BodyHandlers.ofLines());
        if (resp.statusCode() != 200) {
            throw new IOException("sub falhou (" + resp.statusCode() + ")");
        }
        resp.body()
                .filter(json -> json.contains("\"data\""))
                .forEach(json -> onMessage.accept(fromMultibase(extractField(json, "data"))));
    }

    static String extractField(String json, String field) {
        String key = "\"" + field + "\":\"";
        int start = json.indexOf(key);
        if (start < 0) {
            throw new IllegalArgumentException("Campo '" + field + "' ausente: " + json);
        }
        start += key.length();
        return json.substring(start, json.indexOf('"', start));
    }
}

package rep02;

import java.io.*;
import java.net.*;
import java.nio.charset.StandardCharsets;
import java.util.TreeMap;

public class PartitionReplica {
    private final String replicaId;
    private final int k;
    private final RecordFile file;
    private long seq;

    private String leaderHost = null;
    private int leaderPort = -1;
    private final TreeMap<Long, SensorRecord> buffered = new TreeMap<>();
    private boolean recovering = false;

    public PartitionReplica(String replicaId, int k, String api) throws Exception {
        this.replicaId = replicaId;
        this.k = k;
        
        this.file = new RecordFile("replica-" + replicaId + "-p" + k + ".txt");
        this.seq = file.lastSeq();

        IpfsPubSub pubsub = new IpfsPubSub(api);

        System.out.println("Réplica " + replicaId + " (p" + k + ") iniciada. Último seq local: " + seq);

        new Thread(this::initialRecovery).start();

        pubsub.subscribe(Wire.topicData(k), msg -> {
            try {
                String trimmed = msg != null ? msg.trim() : "";
                if (!trimmed.startsWith("REG")) {
                    System.out.println("REJEITADO; Mensagem inválida: " + trimmed);
                    return;
                }

                String[] parts = trimmed.split(";", 4);
                if (parts.length < 4) {
                    System.out.println("REJEITADO; Formato REG incompleto: " + trimmed);
                    return;
                }

                String host = parts[1];
                int port;
                try {
                    port = Integer.parseInt(parts[2]);
                } catch (NumberFormatException e) {
                    System.out.println("REJEITADO; Porto inválido no REG: " + trimmed);
                    return;
                }

                synchronized (this) {
                    leaderHost = host;
                    leaderPort = port;
                }

                SensorRecord r;
                try {
                    r = SensorRecord.fromLine(parts[3]);
                } catch (Exception e) {
                    System.out.println("REJEITADO; Registo de sensor corrompido: " + trimmed);
                    return;
                }

                long rSeq = r.getSeq();

                synchronized (this) {
                    if (rSeq <= seq) {
                        System.out.println("DUPLICADO; Registo " + rSeq + " já processado.");
                        return;
                    }

                    if (rSeq == seq + 1) {
                        file.append(r);
                        seq = r.getSeq();
                        System.out.println("p" + k + " Sincronizado (REG): " + r.toLine());

                        while (buffered.containsKey(seq + 1)) {
                            SensorRecord nextR = buffered.remove(seq + 1);
                            file.append(nextR);
                            seq = nextR.getSeq();
                            System.out.println("p" + k + " Sincronizado (Buffer): " + nextR.toLine());
                        }
                    } else {
                        if (!buffered.containsKey(rSeq)) {
                            buffered.put(rSeq, r);
                            System.out.println("EM ESPERA; Retido seq " + rSeq + ". Total retidos: " + buffered.size());
                        }

                        // Chamada direta e síncrona na thread de subscrição quando ultrapassa 10 retidos[cite: 10]
                        if (buffered.size() > 10) {
                            recoverFromLeader();
                        }
                    }
                }
            } catch (Exception e) {
                System.out.println("REJEITADO; Erro ao processar dados: " + e.getMessage());
            }
        });
    }

    private void initialRecovery() {
        int tentativas = 0;
        while (leaderPort == -1 && tentativas < 5) {
            try {
                Thread.sleep(1000);
            } catch (InterruptedException ignored) {}
            tentativas++;
        }
        if (leaderPort != -1) {
            recoverFromLeader();
        }
    }

    private void recoverFromLeader() {
        synchronized (this) {
            if (recovering) return;
            if (leaderPort == -1 || leaderHost == null) return;
            recovering = true;
        }

        long from;
        long to;
        String host;
        int port;

        synchronized (this) {
            from = seq + 1;
            if (buffered.isEmpty()) {
                to = seq + 5000; 
            } else {
                to = buffered.firstKey() - 1;
            }

            if (from > to) {
                synchronized (this) {
                    recovering = false;
                }
                return;
            }

            host = leaderHost;
            port = leaderPort;
        }

        System.out.println("RECUPERAÇÃO: " + from + ".." + to + " pedida a " + host + ":" + port);

        try (Socket socket = new Socket(host, port);
             BufferedReader in = new BufferedReader(new InputStreamReader(socket.getInputStream(), StandardCharsets.UTF_8));
             PrintWriter out = new PrintWriter(new OutputStreamWriter(socket.getOutputStream(), StandardCharsets.UTF_8), true)) {

            socket.setSoTimeout(Wire.T_TCP_MS);
            out.println("PEDIDO " + from + " " + to);

            String line;
            boolean receivedFim = false;
            while ((line = in.readLine()) != null) {
                line = line.trim();
                if (line.equals("FIM")) {
                    receivedFim = true;
                    break;
                }
                if (line.startsWith("ERRO")) {
                    System.out.println("Líder rejeitou recuperação: " + line);
                    break;
                }

                SensorRecord r = SensorRecord.fromLine(line);
                synchronized (this) {
                    if (r.getSeq() > seq) {
                        file.append(r);
                        seq = r.getSeq();
                        buffered.remove(seq);
                        // Mensagem exata exigida pela ficha[cite: 10]
                        System.out.println("RECUPERADO: " + seq);
                        System.out.println("p" + k + " Recuperado TCP: " + r.toLine());
                    }
                }
            }

            if (!receivedFim) {
                System.out.println("Aviso: Recuperação TCP incompleta (sem FIM). Estado retido mantido em memória.");
            } else {
                synchronized (this) {
                    while (buffered.containsKey(seq + 1)) {
                        SensorRecord nextR = buffered.remove(seq + 1);
                        file.append(nextR);
                        seq = nextR.getSeq();
                        System.out.println("p" + k + " Sincronizado pós-TCP (Buffer): " + nextR.toLine());
                    }
                }
            }

        } catch (IOException e) {
            System.out.println("Aviso: Falha na ligação TCP para recuperação (" + e.getMessage() + "). Estado retido mantido.");
        } finally {
            synchronized (this) {
                recovering = false;
            }
        }
    }

package rep02;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.ArrayList;
import java.util.List;



public class RecordFile {
    private final Path path;

    public RecordFile(String fileName) {
        this.path = Paths.get(fileName);
    }

    public synchronized List<SensorRecord> range(long from, long to) throws IOException {
        List<SensorRecord> out = new ArrayList<>();
        if (!Files.exists(path)) {
            return out;
        }
        for (String line : Files.readAllLines(path, StandardCharsets.UTF_8)) {
            if (line.isBlank()) {
                continue;
            }
            SensorRecord r = SensorRecord.fromLine(line);
            if (r.getSeq() >= from && r.getSeq() <= to) {
                out.add(r);
            }
        }
        return out;
    }

    // acrescenta uma linha ao fim do ficheiro (cria-o se não existir)
    public synchronized void append(SensorRecord r) throws IOException {
        Files.writeString(path, r.toLine() + System.lineSeparator(), StandardCharsets.UTF_8,
                StandardOpenOption.CREATE, StandardOpenOption.APPEND);
    }

    // número de sequência do último registo guardado (0 se o ficheiro não existir)
    public synchronized long lastSeq() throws IOException {
        if (!Files.exists(path)) {
            return 0;
        }
        long last = 0;
        for (String line : Files.readAllLines(path, StandardCharsets.UTF_8)) {
            if (!line.isBlank()) {
                last = SensorRecord.fromLine(line).getSeq();
            }
        }
        return last;
    }
}

public class SensorRecord {

    private final String sensorId;
    private final double temperature;
    private final String timestamp;

    public SensorRecord(long seq, String sensorId, double temperature, String timestamp) {
        this.seq = seq;
        this.sensorId = sensorId;
        this.temperature = temperature;
        this.timestamp = timestamp;
    }

    public static SensorRecord now(long seq, String sensorId, double temperature) {
        return new SensorRecord(seq, sensorId, temperature, LocalDateTime.now().withNano(0).toString());

    public String toLine() {
        return seq + ";" + sensorId + ";" + temperature + ";" + timestamp;
    }

    public static SensorRecord fromLine(String line) {
        String[] f = line.trim().split(";");
        if (f.length != 4) {
            throw new IllegalArgumentException("Registo mal formado: " + line);
        }
        return new SensorRecord(Long.parseLong(f[0]), f[1], Double.parseDouble(f[2]), f[3]);
    }

    public long getSeq() { return seq; }
    public String getSensorId() { return sensorId; }
    public double getTemperature() { return temperature; }
    public String getTimestamp() { return timestamp; }
}

    public static String topicData(int k) { return "rep02-p" + k; }

    public static String in(String sensor, double temp) { return "IN;" + sensor + ";" + temp; }

    // REG;<endereço>;<porto>;<seq>;<sensor>;<temperatura>;<instante>
    public static String reg(String host, int port, SensorRecord r) {
        return "REG;" + host + ";" + port + ";" + r.toLine();
    }

    // tipo da mensagem (IN, REG); lança IllegalArgumentException se não houver tipo
    public static String type(String msg) {
        String m = msg.trim();
        int i = m.indexOf(';');
        if (i <= 0) {
            throw new IllegalArgumentException("Mensagem sem tipo: " + msg);
        }
        return m.substring(0, i);
    }

    // conteúdo da mensagem sem o tipo
    public static String body(String msg) {
        String m = msg.trim();
        return m.substring(m.indexOf(';') + 1);
    }

    // divide uma mensagem REG em { endereço, porto, linha do registo }
    public static String[] regParts(String msg) {
        String[] f = msg.trim().split(";", 4);
        if (f.length != 4 || !f[0].equals("REG")) {
            throw new IllegalArgumentException("REG mal formado: " + msg);
        }
        return new String[] { f[1], f[2], f[3] };
    }

<section>
  <h2><span class="n">2</span>Visão geral da arquitetura</h2>
  <p>O fluxo de dados organiza-se em três troços. O <b>produtor</b> determina a partição de cada leitura a partir da chave de partição (o identificador do sensor) e publica-a no tópico de entrada correspondente. O <b>líder</b> da partição atribui-lhe um número de sequência, regista-a no seu ficheiro local e, em seguida, propaga-a no tópico de dados. As <b>réplicas</b> responsáveis por essa partição aplicam os registos segundo a ordem definida pelo líder. A recuperação de registos perdidos constitui a única interação fora do PubSub: realiza-se por comunicação direta, através de uma ligação TCP da réplica para o endereço e porto anunciados pelo líder na mensagem <code>REG</code> mais recente.</p>

<section>
  <h2><span class="n">3</span>Papéis e conhecimento mútuo dos processos</h2>
  <p>A caracterização de cada papel assenta no conhecimento que o processo detém sobre os restantes. Nos troços mediados pelo PubSub preserva-se o desacoplamento no espaço; apenas o mecanismo de recuperação o abandona, introduzindo acoplamento no espaço e no tempo entre a réplica e o líder.</p>
  <div class="roles">
    <div class="role">
      <span class="proj">rep02-produtor · Producer</span>
      <h3>Produtor</h3>
      <p>Constitui o ponto de entrada das leituras no sistema. Determina a partição de cada leitura através da chave de partição e publica a mensagem <code>IN</code> no tópico de entrada respetivo.</p>
      <dl><dt>Conhece</dt><dd>o número de partições e a função de partição</dd><dt>Desconhece</dt><dd>a identidade dos líderes e das réplicas</dd><dt>Limitação</dt><dd>não numera as leituras; uma perda neste troço não é detetável</dd></dl>
    </div>
    <div class="role">
      <span class="proj">rep02-lider · PartitionLeader</span>
      <h3>Líder da partição k</h3>
      <p>Define a ordem dos registos da partição k. Atribui o número de sequência, escreve o registo localmente e só depois o propaga. O ficheiro local constitui a fonte da recuperação.</p>
      <dl><dt>Conhece</dt><dd>k, N, o endereço anunciado e o porto de recuperação</dd><dt>Desconhece</dt><dd>o número e a identidade das réplicas</dd><dt>Verifica</dt><dd>que <code>of(sensor) == k</code>; caso contrário, rejeita a leitura</dd></dl>
    </div>
    <div class="role">
      <span class="proj">rep02-replica · PartitionReplica</span>
      <h3>Réplica n</h3>
      <p>Mantém as partições que lhe foram atribuídas, cada uma num ficheiro próprio. Aplica os registos por ordem, retém em memória os que chegam antecipadamente e solicita ao líder os que considera perdidos.</p>
      <dl><dt>Conhece</dt><dd>o seu identificador e as partições atribuídas</dd><dt>Obtém</dt><dd>o endereço e o porto do líder a partir de cada <code>REG</code></dd><dt>Decide</dt><dd>que um registo se perdeu quando os retidos excedem 10</dd></dl>
    </div>
    <div class="role">
      <span class="proj">rep02-replica · PartitionCheck</span>
      <h3>Verificador</h3>
      <p>Ferramenta de análise a posteriori. Lê os ficheiros <code>lider-p*.txt</code> e <code>replica-*-p*.txt</code> e verifica a correção da partição, a continuidade das sequências, o número de réplicas por partição e a relação de igualdade ou de prefixo com o ficheiro do líder.</p>
      <dl><dt>Execução</dt><dd>no final de cada ensaio</dd><dt>Comunicação</dt><dd>nenhuma; opera apenas sobre ficheiros</dd></dl>
    </div>
    <div class="role">
      <span class="proj">executável ipfs (Kubo)</span>
      <h3>Daemon IPFS</h3>
      <p>Intermediário da comunicação indireta, acedido pela API HTTP (<code>/api/v0</code>, porto 5001) através da classe <code>IpfsPubSub</code>. Oferece entrega <em>best-effort</em>, sujeita a perdas e a reordenação.</p>
      <dl><dt>Assegura</dt><dd>desacoplamento no espaço e no tempo</dd><dt>Não assegura</dt><dd>entrega nem ordem</dd></dl>
    </div>
  </div>
</section>

<section>
  <h2><span class="n">5</span>Organização em projetos e classes</h2>
  <p>O sistema distribui-se por três projetos, que simulam máquinas distintas; todas as classes pertencem ao pacote <code>rep02</code>. As classes herdadas da COMINDIR01 são copiadas sem alteração além do pacote, exceto <code>RecordFile</code>, que recebe o método <code>range</code>.</p>

<section>
  <h2><span class="n">6</span>Concorrência e estado partilhado</h2>
  <div class="threads">
    <div class="lane"><h3>PartitionLeader</h3>
      <div class="th"><b>Atividade de entrada</b> (thread dedicada)<br>Bloqueia em <code>subscribe(topicIn(k))</code>. Escreve <code>ultimoSeq</code> e o ficheiro do líder.</div>
      <div class="th"><b>Servidor de recuperação</b> (thread principal)<br>Bloqueia em <code>accept()</code> e, posteriormente, em <code>readLine()</code>. Lê <code>ultimoSeq</code>, para limitar o intervalo, e o ficheiro, através de <code>range</code>. As ligações são atendidas sequencialmente.</div>
      <div class="shared"><b>Estado partilhado:</b> <code>ultimoSeq</code> e <code>lider-p&lt;k&gt;.txt</code>. A atribuição do número de sequência e o <code>append</code> devem integrar a mesma secção sincronizada; caso contrário, o servidor poderia observar um número de sequência ainda não persistido.</div>
    </div>
    <div class="lane"><h3>PartitionReplica</h3>
      <div class="th"><b>Uma thread por partição</b><br>Bloqueia em <code>subscribe(topicData(k))</code>. A recuperação é executada nesta mesma thread e bloqueia em <code>connect</code> e <code>readLine</code>, no máximo <code>T_TCP_MS</code>.</div>
      <div class="th"><b>Thread principal</b><br>Inicia as threads de subscrição e aguarda a sua conclusão.</div>
      <div class="shared"><b>Estado partilhado:</b> reduzido, dado que cada partição dispõe de estado e ficheiro próprios. Exigem atenção o estado global da simulação (o descarte já efetuado e o gerador aleatório) e a escrita na consola. A recuperação de uma partição não interrompe a receção das restantes; durante esse período, as mensagens da partição em recuperação permanecem por ler na resposta HTTP do pedido de subscrição.</div>
    </div>
    <div class="lane"><h3>Producer</h3>
      <div class="th"><b>Thread principal</b><br>Interpreta a consola e publica; o comando <code>rajada</code> publica em ciclo, à máxima cadência possível.</div>
      <div class="shared" style="background:var(--code-bg)"><b>Sem estado partilhado.</b> O produtor não conserva as leituras enviadas, pelo que não tem capacidade de as retransmitir.</div>
    </div>
  </div>
</section>

<section>
  <h2><span class="n">7</span>Regra de aplicação na réplica</h2>
  <p>A regra aplica-se independentemente a cada partição, tanto aos registos <code>REG</code> recebidos por PubSub como aos registos obtidos por recuperação.</p>
  <div class="rules">
    <div class="rule r-ok"><span class="cond">seq = último + 1</span><span>O registo é escrito; seguidamente, escrevem-se os registos retidos que passaram a estar em sequência (enquanto <code>retidos.firstKey() = último + 1</code>).</span></div>
    <div class="rule r-dup"><span class="cond">seq ≤ último, ou já retido</span><span>O registo é considerado duplicado e ignorado. Esta condição impede a repetição de registos recebidos simultaneamente por PubSub e por TCP.</span></div>
    <div class="rule r-hold"><span class="cond">seq &gt; último + 1</span><span>O registo é retido em memória. Se o número de retidos exceder <code>LIMITE_RETIDOS</code>, solicita-se o intervalo de <code>último + 1</code> a <code>retidos.firstKey() − 1</code>; cada registo recebido é aplicado segundo a mesma regra e, por fim, escrevem-se os retidos contíguos.</span></div>
    <div class="rule r-fail"><span class="cond">falha na recuperação</span><span>Em caso de ligação recusada, tempo esgotado ou interrupção antes de <code>FIM</code>, mantêm-se os registos já escritos e os retidos em memória. A recuperação é novamente tentada na receção do registo seguinte da partição, uma vez que a condição persiste.</span></div>
  </div>
  <p class="note">Uma réplica iniciada com o ficheiro vazio retém todos os registos recebidos e, à décima primeira mensagem retida, obtém numa única ligação todo o histórico anterior da partição. A perda do último registo de uma série não é recuperada, porquanto não chegam os dez registos posteriores necessários para desencadear o pedido.</p>
</section>

<section>
  <h2><span class="n">10</span>Limitações da arquitetura</h2>
  <ul class="limits">
    <li>As perdas no troço produtor → líder não são detetadas, por ausência de numeração nesse troço.</li>
    <li>A perda dos últimos registos de uma série só é detetada após a receção de mais de dez registos posteriores.</li>
    <li>A falha de um líder impede a escrita na respetiva partição e a recuperação das suas réplicas; o sistema não contempla eleição de líder nem consenso.</li>
    <li>Não existe ordem entre partições: cada partição possui uma sequência própria e os respetivos números não são comparáveis.</li>
    <li>Uma chave muito frequente (80 % das leituras de S01) concentra a carga numa única partição.</li>
    <li>A alteração de N reatribui a maioria dos sensores a outra partição, invalidando a correspondência dos ficheiros existentes.</li>
    <li>Não há verificação de origem: um processo arbitrário pode publicar um <code>REG</code> com um endereço falso e desviar os pedidos de recuperação.</li>
  </ul>
</section>

  function jhash(s) { let h = 0; for (const c of s) h = (Math.imul(h, 31) + c.charCodeAt(0)) | 0; return h; }

  function poly(pts) {
    const cum = [0];
    for (let i = 1; i < pts.length; i++) cum.push(cum[i - 1] + Math.hypot(pts[i][0] - pts[i - 1][0], pts[i][1] - pts[i - 1][1]));
    return { pts, cum, total: cum[cum.length - 1] };
  }
  function pointAt(p, t) {
    const d = Math.max(0, Math.min(1, t)) * p.total;
    let i = 1;
    while (i < p.cum.length - 1 && p.cum[i] < d) i++;
    const seg = p.cum[i] - p.cum[i - 1] || 1, f = (d - p.cum[i - 1]) / seg;
    const a = p.pts[i - 1], b = p.pts[i];
    return [a[0] + (b[0] - a[0]) * f, a[1] + (b[1] - a[1]) * f];
  }
  function arcPts(x1, y1, x2, y2, cx, cy) {
    const out = [];
    for (let i = 0; i <= 28; i++) {
      const t = i / 28, u = 1 - t;
      out.push([u * u * x1 + 2 * u * t * cx + t * t * x2, u * u * y1 + 2 * u * t * cy + t * t * y2]);
    }
    return out;
  }

        continue;

  // réplica
  function replicaReceive(r, k, seq) {

    P.addr = HOST + ':' + (PORT + k);

  function apply(r, k, seq, src) {
    const P = S.reps[r].parts[k];
    if (seq <= P.last || P.retained.has(seq)) { if (src === 'pubsub') log(`R${r + 1}/p${k}: DUPLICADO ${seq}`, 'q'); return; }
    if (seq === P.last + 1) {
      P.last = seq;
      if (src === 'tcp') log(`R${r + 1}/p${k}: RECUPERADO ${seq}`, 'tcp');
      drain(P);
      return;
    }
    P.retained.add(seq);
    log(`R${r + 1}/p${k}: EM ESPERA ${seq} (falta ${P.last + 1}) · retidos ${P.retained.size}`);
    if (P.retained.size > LIM && !P.recovering && src === 'pubsub') recover(r, k);
  }
  function drain(P) { while (P.retained.has(P.last + 1)) { P.retained.delete(P.last + 1); P.last++; } }

  function recover(r, k) {
    const P = S.reps[r].parts[k];
    const a = P.last + 1, b = Math.min(...P.retained) - 1;
    P.recovering = true;
    const id = ++P.recId;

    const alive = () => S.reps[r].online && P.recId === id;
    spawn(there, 0.8, 'var(--tcp)', `PEDIDO ${a} ${b}`, () => {
      if (!alive()) return;
      const recs = S.leaders[k].file.filter(x => x.seq >= a && x.seq <= b);
      log(`Líder p${k}: RECUPERAÇÃO ${a}..${b} para R${r + 1} (${recs.length} registos)`, 'tcp');
      ui.leaders[k].flash = 0.5;
      say(`<b>Resposta do líder ${k}:</b> lê os registos ${a} a ${b} de <code>lider-p${k}.txt</code> e envia-os, por ordem, na mesma ligação, seguidos de <code>FIM</code>. As mensagens do PubSub que chegam entretanto à réplica ficam em fila.`);
      const step = Math.min(0.14, 2.4 / Math.max(1, recs.length));
      recs.forEach((x, i) => schedule(i * step, () => {
        if (!alive()) return;
        spawn(back, 0.7, 'var(--tcp)', String(x.seq), () => { if (alive()) apply(r, k, x.seq, 'tcp'); }, { square: true });
      }));
      schedule(recs.length * step + 0.05, () => {
        if (!alive()) return;
        spawn(back, 0.7, 'var(--tcp)', 'FIM', () => { if (alive()) finish(r, k); }, { r: 4 });
      });
    });
  }
  function finish(r, k) {
    const P = S.reps[r].parts[k];
    P.recovering = false;
    if (P.arc) { P.arc.remove(); P.arc = null; }
    drain(P);
    const q = P.queue; P.queue = [];
    log(`R${r + 1}/p${k}: FIM recebido; último ${P.last}; ${q.length} mensagens em fila aplicadas`, 'ok');
    say(`A réplica ${r + 1} fechou a ligação, escreveu os retidos que ficaram em sequência e aplicou as ${q.length} mensagens que aguardavam em fila. A partição ${k} está no registo ${P.last}.`);
    for (const s of q) {
      if (P.recovering) P.queue.push(s); else apply(r, k, s, 'pubsub');
    }
  }

  function toggleReplica(r) {
    const R = S.reps[r];
    R.online = !R.online;
    if (!R.online) {
      for (const k in R.parts) {
        const P = R.parts[k];
        P.retained.clear(); P.queue = []; P.recovering = false; P.recId++;
        if (P.arc) { P.arc.remove(); P.arc = null; }
      }
      log(`Réplica ${r + 1} parada: a memória (retidos e fila) perde-se; os ficheiros mantêm-se`, 'bad');
      say(`A réplica ${r + 1} está parada. Não recebe as mensagens publicadas: o PubSub não as guarda.`);
    } else {
      const parts = Object.entries(R.parts).map(([k, P]) => `p${k}=${P.last}`).join(', ');
      log(`Réplica ${r + 1} arrancada: lastSeq() ${parts}`, 'ok');
      say(`A réplica ${r + 1} voltou a arrancar com os ficheiros em ${parts}. Vai reter o que receber e recuperar por TCP quando ultrapassar ${LIM} retidos.`);
    }
    syncButtons();
  }

function drain(r,who,quiet){

async function onReg(r,rec,id){
  const who='R'+r.id,seq=rec.seq,q=S.quiet;
  if(r.descartar===seq){r.descartar=null;r.drops++;log(who,'DESCARTADO (simulação): '+seq,'warn');setRule(`R${r.id} · REG ${seq}\n→ descartado antes de ser aplicado\n  (-Ddescartar=${seq})`);return;}
  if(r.perda&&rndLoss()<r.perda){r.drops++;log(who,'DESCARTADO (simulação): '+seq,'warn');return;}
  r.addr=rec.addr+':'+rec.port;
  if(seq<=r.last||r.held.has(seq)){log(who,'DUPLICADO: '+seq,'warn');setRule(`R${r.id} · REG ${seq}\nseq ≤ último (${r.last}) ou já retido\n→ ignora (duplicado)`);}
  else if(seq===r.last+1){append(r.file,rec.line);r.last=seq;if(!q)log(who,'Aplicado: '+rec.line);
    const before=r.held.size;drain(r,who,q);
    setRule(`R${r.id} · REG ${seq}\nseq = último + 1\n→ escreve no ficheiro`+(before?`\n→ e escreve ${before-r.held.size} retidos`:''));}
  else{r.held.set(seq,rec.line);log(who,`EM ESPERA: ${seq} (falta o registo ${r.last+1})`,'warn');
    setRule(`R${r.id} · REG ${seq}\nseq > último + 1 (${r.last+1})\n→ retém em memória\n  retidos = ${r.held.size}${r.held.size>LIMITE?' > ':' ≤ '}LIMITE_RETIDOS (${LIMITE})`);}
  if(r.held.size>LIMITE)await recover(r,id);
}
async function recover(r,id){
  const i=r.id-1,who='R'+r.id,L=S.L;
  const from=r.last+1,to=Math.min(...r.held.keys())-1;
  r.busy=true;r.conns++;r.stat='a recuperar por TCP';
  pTCP[i].classList.add('on');
  log(who,`RECUPERAÇÃO: ${from}..${to} pedida a ${r.addr}`,'tcp');
  setRule(`R${r.id} · retidos = ${r.held.size} > ${LIMITE}\n→ pede de último + 1 (${from})\n  até menor retido − 1 (${to})`);
  let tr=`R${r.id} → ${r.addr}\nconnect(…, T_TCP_MS)\n→ PEDIDO ${from} ${to}\n`;setTcp(tr);
  try{
    await fly(pTCP[i],`PEDIDO ${from} ${to}`,id,{kind:'tcpq',dur:1000});
    while(L.busy)await sleep(60,id);           // o líder atende uma ligação de cada vez
    L.busy=true;L.srv=`a atender R${r.id}`;
    const top=Math.min(to,L.seq);
    const recs=L.file.filter(x=>{const s=seqOf(x.s);return s>=from&&s<=top;});
    log('L',`RECUPERAÇÃO: ${from}..${top} para 127.0.0.1:${52000+r.id*37+r.conns} (${recs.length} registos)`,'tcp');
    const applyOne=x=>{const s=seqOf(x.s);if(s===r.last+1){append(r.file,x.s,true);r.last=s;r.recovered++;log(who,'RECUPERADO: '+s,'tcp');}};
    if(recs.length<=4){
      for(const x of recs){await fly(pTCP[i],String(seqOf(x.s)),id,{kind:'tcpr',dur:650,reverse:true});tr+='← '+x.s+'\n';setTcp(tr);applyOne(x);}
    }else{
      await fly(pTCP[i],recs.length+' registos',id,{kind:'tcpr',dur:1100,reverse:true});
      recs.slice(0,3).forEach(x=>tr+='← '+x.s+'\n');tr+=`← … (${recs.length} linhas no total)\n`;setTcp(tr);
      recs.forEach(applyOne);
    }
    await fly(pTCP[i],'FIM',id,{kind:'tcpr',dur:550,reverse:true});
    tr+='← FIM   (o líder fecha a ligação)';setTcp(tr);
    L.busy=false;L.srv='accept() à espera';
    const before=r.held.size;drain(r,who,S.quiet);
    setRule(`R${r.id} · recuperação concluída\n→ ${recs.length} registos por TCP\n→ ${before-r.held.size} retidos escritos\n  retidos agora = ${r.held.size}`);
  }finally{pTCP[i].classList.remove('on');r.busy=false;if(r.sub)r.stat='subscribe() bloqueado';}
}

async function onReg(r,k,rec,id){
  const who='R'+r.id,p=r.ps[k],seq=rec.seq,q=S.quiet;
  p.addr=rec.addr+':'+rec.port;
  if(seq<=p.last||p.held.has(seq))log(who,`p${k} DUPLICADO: ${seq}`,'warn');
  else if(seq===p.last+1){append(p.file,rec.line);p.last=seq;if(!q)log(who,`p${k} aplicado: `+rec.line);drain(p,who,k,q);}

}
async function recover(r,k,id){
  const i=r.id-1,who='R'+r.id,p=r.ps[k],L=S.L[k];
  const from=p.last+1,to=Math.min(...p.held.keys())-1;
  p.busy=true;p.conns++;
  const path=tcpPath(i,k);
  log(who,`p${k} RECUPERAÇÃO: ${from}..${to} pedida a ${p.addr}`,'tcp');
  setRule(`R${r.id} · partição ${k}\nretidos = ${p.held.size} > ${LIMITE}\n→ PEDIDO ${from} ${to} a ${p.addr}\n\nas outras partições de R${r.id}\ncontinuam a ser lidas`);
  try{
    await fly(path,`PEDIDO ${from} ${to}`,id,{kind:'tcpq',dur:1000,color:PC(k)});
    while(L.busy)await sleep(60,id);
    L.busy=true;
    const top=Math.min(to,L.seq);
    const recs=L.file.filter(x=>{const s=seqOf(x.s);return s>=from&&s<=top;});
    log('L'+k,`RECUPERAÇÃO: ${from}..${top} para 127.0.0.1:${52000+r.id*37+p.conns} (${recs.length} registos)`,'tcp');
    await fly(path,recs.length+(recs.length===1?' registo':' registos'),id,{kind:'tcpr',dur:1000,reverse:true,color:PC(k)});
    recs.forEach(x=>{const s=seqOf(x.s);if(s===p.last+1){append(p.file,x.s,true);p.last=s;p.recovered++;if(!S.quiet)log(who,`p${k} RECUPERADO: ${s}`,'tcp');}});
    if(S.quiet&&recs.length)log(who,`p${k} RECUPERADO: ${seqOf(recs[0].s)} … ${seqOf(recs[recs.length-1].s)}`,'tcp');
    await fly(path,'FIM',id,{kind:'tcpr',dur:500,reverse:true,color:PC(k)});
    L.busy=false;
    drain(p,who,k,S.quiet);
  }finally{path.remove();p.busy=false;}
}
